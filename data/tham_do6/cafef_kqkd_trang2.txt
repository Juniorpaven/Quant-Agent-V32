        mergeInsuranceKqkdMongoDetail(apiPeriods, mongoPeriods) {
            if (!apiPeriods || !mongoPeriods || !mongoPeriods.length) {
                return apiPeriods;
            }

            var normalizeTime = function (timeBctc) {
                var time = String((timeBctc && timeBctc.time) || '').trim();
                var content = String((timeBctc && timeBctc.content) || '').trim();
                if (time && content && time.indexOf(content) < 0 && !/\d{4}/.test(time)) {
                    return (time + '-' + content).toUpperCase();
                }
                return (time || content).toUpperCase();
            };

            var getPeriodYear = function (period) {
                if (period && period.year) {
                    return period.year;
                }
                if (period && period.Year) {
                    return period.Year;
                }
                var time = normalizeTime(period);
                var match = time.match(/(\d{4})/);
                return match ? parseInt(match[1], 10) : 0;
            };

            var getPeriodQuarter = function (period) {
                if (period && period.quater) {
                    return period.quater;
                }
                if (period && period.quarter) {
                    return period.quarter;
                }
                if (period && period.Quarter) {
                    return period.Quarter;
                }
                var time = normalizeTime(period);
                var match = time.match(/Q(\d)/i);
                return match ? parseInt(match[1], 10) : 0;
            };

            var findMongoPeriod = function (apiPeriod) {
                var apiKey = normalizeTime(apiPeriod);
                var apiYear = getPeriodYear(apiPeriod);
                var apiQuarter = getPeriodQuarter(apiPeriod);
                return mongoPeriods.find(function (mongoPeriod) {
                    if (normalizeTime(mongoPeriod) === apiKey) {
                        return true;
                    }
                    var mongoYear = getPeriodYear(mongoPeriod);
                    var mongoQuarter = getPeriodQuarter(mongoPeriod);
                    return apiYear > 0 && apiQuarter > 0
                        && mongoYear === apiYear
                        && mongoQuarter === apiQuarter;
                });
            };

            apiPeriods.forEach(function (apiPeriod) {
                var mongoPeriod = findMongoPeriod(apiPeriod);
                if (!mongoPeriod || !mongoPeriod.Data || !mongoPeriod.Data.length) {
                    return;
                }

                if (mongoPeriod.HasLctt === true) {
                    apiPeriod.hasLctt = true;
                } else if (mongoPeriod.HasLctt === false) {
                    apiPeriod.hasLctt = false;
                }

                if (mongoPeriod.UsesOldForm === true) {
                    apiPeriod.usesOldForm = true;
                } else if (mongoPeriod.UsesOldForm === false) {
                    apiPeriod.usesOldForm = false;
                }

                if (!apiPeriod.data) {
                    apiPeriod.data = [];
                }

                var existingCodes = {};
                apiPeriod.data.forEach(function (item) {
                    existingCodes[String(item.code || item.MaSo || '')] = true;
                });

                mongoPeriod.Data.forEach(function (mongoItem) {
                    var code = String(mongoItem.Code || mongoItem.code || '');
                    if (!code || existingCodes[code]) {
                        return;
                    }
                    apiPeriod.data.push({
                        code: code,
                        value: mongoItem.Value != null ? mongoItem.Value : mongoItem.value
                    });
                    existingCodes[code] = true;
                });
            });

            return apiPeriods;
        },

        async enrichInsuranceKqkdFromMongo(symbol, pageIndex, typeTime, data) {

            try {
                var mongoUrl = `/du-lieu/Ajax/PageNew/GetInsuranceKqkdDetail.ashx?Symbol=${symbol}&pageIndex=${pageIndex}&pageSize=4&TypeTime=${typeTime}`;
                var mongoRes = await fetch(mongoUrl);
                if (!mongoRes.ok) {
                    return;
                }
                var mongoJson = await mongoRes.json();
                if (mongoJson && mongoJson.Success && mongoJson.Data && mongoJson.Data.length) {
                    this.mergeInsuranceKqkdMongoDetail(data.value.data, mongoJson.Data);
                } else if (mongoJson && mongoJson.Success === false) {
                    console.warn('Insurance KQKD mongo enrich failed:', mongoJson.Message || 'unknown');
                }
            } catch (mongoErr) {
                console.warn('Insurance KQKD mongo enrich skipped:', mongoErr);
            }
        },

                if (wrapper) {
                    wrapper.innerHTML = '<div style="display:flex;padding:12px;justify-content:center;">Không tải được dữ liệu</div>';
                }
            }

        bindExportExcel: function () {
            var btn = document.querySelector('.chose_time_unit_finacial_right-item');
            if (btn) {
                btn.style.cursor = 'pointer';
                btn.addEventListener('click', () => {
                    this.exportToExcel();
                });
            }
        },
        exportToExcel: function () {
            var wrapperTables = Array.from(document.querySelectorAll('.table-wrapper-item table'));
            
            var tables = [...wrapperTables];
            if (!tables.length) {
                alert("Không có dữ liệu để xuất Excel.");
                return;
            }
            
            var excelHtml = `
            <html xmlns:o="urn:schemas-microsoft-com:office:office" xmlns:x="urn:schemas-microsoft-com:office:excel" xmlns="http://www.w3.org/TR/REC-html40">
            <head>
                <meta charset="utf-8" />
                <!--[if gte mso 9]>
                <xml>
                    <x:ExcelWorkbook>
                        <x:ExcelWorksheets>
                            <x:ExcelWorksheet>
                                <x:Name>BCTC_Tom_Tat</x:Name>
                                <x:WorksheetOptions>
                                    <x:DisplayGridlines/>
                                </x:WorksheetOptions>
                            </x:ExcelWorksheet>
                        </x:ExcelWorksheets>
                    </x:ExcelWorkbook>
                </xml>
                <![endif]-->
                <style>
                    table { border-collapse: collapse; margin-bottom: 20px; font-family: Arial, sans-serif; }
                    th, td { border: 1px solid #ccc; padding: 8px; text-align: left; }
                    th { background-color: #f2f2f2; font-weight: bold; }
                    .title-item { color: #C32022; font-size: 16px; font-weight: bold; }
                    .level1-item { font-weight: bold; background-color: #F8F8F8; }
                    .level2-item { font-weight: normal; }
                    .level3-item { font-weight: normal; }
                    .number-item { text-align: right; }
                </style>
            </head>
            <body>
            `;

            tables.forEach(table => {
                excelHtml += '<table>';
                
                // Process thead
                var thead = table.querySelector('thead');
                if (thead) {
                    excelHtml += '<thead>';
                    var rows = thead.querySelectorAll('tr');
                    rows.forEach(row => {
                        excelHtml += '<tr>';
                        var ths = row.querySelectorAll('th');
                        ths.forEach((th, index) => {
                            // Skip next/prev pagination columns and growth column
                            if (th.classList.contains('click_page-item') || th.textContent.includes('Tăng trưởng')) {
                                return;
                            }
                            excelHtml += `<th>${th.innerText.trim()}</th>`;
                        });
                        excelHtml += '</tr>';
                    });
                    excelHtml += '</thead>';
                }
                
                // Process tbody
                var tbody = table.querySelector('tbody');
                if (tbody) {
                    excelHtml += '<tbody>';
                    var rows = tbody.querySelectorAll('tr');
                    rows.forEach(row => {
                        var levelClass = '';
                        if (row.classList.contains('level1-item')) levelClass = ' class="level1-item"';
                        else if (row.classList.contains('level2-item')) levelClass = ' class="level2-item"';
                        else if (row.classList.contains('level3-item')) levelClass = ' class="level3-item"';
                        
                        excelHtml += `<tr${levelClass}>`;
                        var tds = row.querySelectorAll('td');
                        tds.forEach((td, index) => {
                            // Skip next/prev pagination columns and growth column (bars)
                            if (td.classList.contains('click_page-item') || td.classList.contains('growth-item')) {
                                return;
                            }
                            
                            var isNumber = td.classList.contains('number-item');
                            var styleAttr = '';
                            if (isNumber) {
                                styleAttr = ' class="number-item"';
                                var isNegative = td.style.color === 'rgb(195, 32, 34)' || td.getAttribute('style')?.includes('#C32022');
                                if (isNegative) {
                                    styleAttr = ' class="number-item" style="color: #C32022;"';
                                }
                            }
                            excelHtml += `<td${styleAttr}>${td.innerText.trim()}</td>`;
                        });
                        excelHtml += '</tr>';
                    });
                    excelHtml += '</tbody>';
                }
                
                excelHtml += '</table>';
            });

            excelHtml += '</body></html>';

            var symbol = 'fpt'.toUpperCase() || 'BCTC';
            var fileName = `${symbol}_KQKD.xls`;
            var blob = new Blob([excelHtml], { type: 'application/vnd.ms-excel;charset=utf-8;' });
            
            if (window.navigator && window.navigator.msSaveBlob) { // IE 10+
                window.navigator.msSaveBlob(blob, fileName);
            } else {
                var link = document.createElement("a");
                if (link.download !== undefined) {
                    var url = URL.createObjectURL(blob);
                    link.setAttribute("href", url);
                    link.setAttribute("download", fileName);
                    link.style.visibility = 'hidden';
                    document.body.appendChild(link);
                    link.click();
                    document.body.removeChild(link);
                }
            }
        },
        bindDropdownTime: function () {
            if (!this.selectedBox || !this.dropdown) {
                return;
            }
            this.selectedBox.addEventListener('click', () => {
                this.dropdownUnit.classList.remove('active-item');
                this.dropdown.classList.toggle('active-item');
            });
            this.items.forEach(item => {
                item.addEventListener('click', () => {
                    this.selected.textContent = item.textContent.trim();
                    this.items.forEach(i => {
                        i.classList.remove('active_drop_time-item');
                    });
                    item.classList.add('active_drop_time-item');
                    this.dropdown.classList.remove('active-item');
                    
                    var typeTime = item.getAttribute('data-time') || 'QUY';
                    this.getDataReportSummary(1, typeTime);
                });
            });
        },
        bindDropdownUnit: function () {
            if (!this.selectedUnitBox || !this.dropdownUnit) {
                return;
            }
            this.selectedUnitBox.addEventListener('click', () => {
                this.dropdown.classList.remove('active-item');
                this.dropdownUnit.classList.toggle('active-item');
            });
            this.unitItems.forEach(item => {
                item.addEventListener('click', () => {
                    this.selectedUnit.textContent = item.textContent.trim();
                    this.unitItems.forEach(i => {
                        i.classList.remove('active_drop_unit-item');
                    });
                    item.classList.add('active_drop_unit-item');
                    this.dropdownUnit.classList.remove('active-item');
                    
                    this.renderDataAll(this.tableDataCS["KQKD"], this.currentPageIndex || 1);
                });
            });
        },
        bindClickOutside: function () {
            document.addEventListener('click', (e) => {
                if (!this.dropdown.contains(e.target)) {
                    this.dropdown.classList.remove('active-item');
                }
                if (!this.dropdownUnit.contains(e.target)) {
                    this.dropdownUnit.classList.remove('active-item');
                }
            });
        }
    };
    TAI_CHINH_Item.init();
</script>

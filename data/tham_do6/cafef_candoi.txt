        async getDataReportSummary(pageIndex = 1, pagesize = 4, type = "ALL", typeTime = "QUY") {
            try {
                if (type !== "ALL") {
                    this.pageIndexes[type] = pageIndex;
                } else {
                    this.pageIndexes = {
                        TN: 1,
                        NV:1
                    };
                }
                const response = await fetch(`https://apiweb.cafef.vn/api/v2/BCTC/GetReportCDKT?symbol=FPT&pageIndex=${pageIndex}&pageSize=${pagesize}&reportType=${type}&TypeTime=${typeTime}`);
                const data = await response.json();
                this.currentData = data;
                this.currentPageIndex = pageIndex;
                this.currentPageSize = pagesize;
                this.currentType = type;
                this.currentTypeTime = typeTime;

                if (type != "ALL") {
                    this.tableData[type] = data;
                    await this.renderDatachild(data, `table_${type}`)
                } else {
                    data.value.templace.forEach(temp => {
                        var singleData = data.value.data.find(x => x.code === temp.code);
                        if (singleData) {
                            this.tableData[temp.code] = {
                                value: {
                                    templace: [temp],
                                    data: [singleData],
                                    count: data.value.count
                                }
                            };
                        }
                    });
                    await this.renderDataAll(data);
                }

                return data;
            } catch (error) {
                console.log(error);
            }

                        else if (row.classList.contains('level3-item')) levelClass = ' class="level3-item"';
                        
                        excelHtml += `<tr${levelClass}>`;
                        var tds = row.querySelectorAll('td');
                        tds.forEach((td, index) => {
                            // Skip next/prev pagination columns and growth column (bars)
                            if (td.classList.contains('click_page-item') || td.classList.contains('growth-item')) {
                                return;
                            }
                            
                            var isNumber = td.classList.contains('number-item');
                            var styleAttr = '';
                            if (isNumber) {
                                styleAttr = ' class="number-item"';
                                var isNegative = td.style.color === 'rgb(195, 32, 34)' || td.getAttribute('style')?.includes('#C32022');
                                if (isNegative) {
                                    styleAttr = ' class="number-item" style="color: #C32022;"';
                                }
                            }
                            excelHtml += `<td${styleAttr}>${td.innerText.trim()}</td>`;
                        });
                        excelHtml += '</tr>';
                    });
                    excelHtml += '</tbody>';
                }
                
                excelHtml += '</table>';
            });

            excelHtml += '</body></html>';

            var symbol = 'fpt'.toUpperCase() || 'BCTC';
            var fileName = `${symbol}_CDKT.xls`;
            var blob = new Blob([excelHtml], { type: 'application/vnd.ms-excel;charset=utf-8;' });

            if (window.navigator && window.navigator.msSaveBlob) { // IE 10+
                window.navigator.msSaveBlob(blob, fileName);
            } else {
                var link = document.createElement("a");
                if (link.download !== undefined) {
                    var url = URL.createObjectURL(blob);
                    link.setAttribute("href", url);
                    link.setAttribute("download", fileName);
                    link.style.visibility = 'hidden';
                    document.body.appendChild(link);
                    link.click();
                    document.body.removeChild(link);
                }
            }
        },
        bindDropdownTime: function () {
            this.selectedBox.addEventListener('click', () => {
                this.dropdownUnit.classList.remove('active-item');
                this.dropdown.classList.toggle('active-item');
            });
            this.items.forEach(item => {
                item.addEventListener('click', () => {
                    this.selected.textContent = item.textContent.trim();
                    this.items.forEach(i => {
                        i.classList.remove('active_drop_time-item');
                    });
                    item.classList.add('active_drop_time-item');
                    this.dropdown.classList.remove('active-item');

                    var typeTime = item.getAttribute('data-time') || 'QUY';
                    this.getDataReportSummary(1, 4, "ALL", typeTime);
                });
            });
        },
        bindDropdownUnit: function () {
            this.selectedUnitBox.addEventListener('click', () => {
                this.dropdown.classList.remove('active-item');
                this.dropdownUnit.classList.toggle('active-item');
            });
            this.unitItems.forEach(item => {
                item.addEventListener('click', () => {
                    this.selectedUnit.textContent = item.textContent.trim();
                    // Update UI active classes
                    this.unitItems.forEach(i => {
                        i.classList.remove('active_drop_unit-item');
                    });
                    item.classList.add('active_drop_unit-item');
                    this.dropdownUnit.classList.remove('active-item');
                    // Re-render all tables with updated unit selection using stored data
                    if (this.currentData) {
                        this.renderDataAll(this.currentData);
                    }
                });
            });
        },
        bindClickOutside: function () {
            document.addEventListener('click', (e) => {
                if (!this.dropdown.contains(e.target)) {
                    this.dropdown.classList.remove('active-item');
                }
                if (!this.dropdownUnit.contains(e.target)) {
                    this.dropdownUnit.classList.remove('active-item');
                }
            });
        }
    };
    TAI_CHINH_Item.init();
</script>
